295 template <typename TargetType = BaseType, typename OriginType>
296 requires(has_mpi_type<OriginType> && has_mpi_type<TargetType>)
297 void get(OriginType *origin_addr, int origin_count, int target_rank, MPI_Aint target_disp = 0, int target_count = -1) const {
298 ASSERT(origin_count >= 0 && target_disp >= 0);
299 target_count = target_count < 0 ? origin_count : target_count;
300 if (has_env) {
301 auto origin_datatype = mpi_type<OriginType>::get();
302 auto target_datatype = mpi_type<TargetType>::get();
303 check_mpi_call(MPI_Get(origin_addr, origin_count, origin_datatype, target_rank, target_disp, target_count, target_datatype, win_), "MPI_Get");
304 } else {
305 std::copy(data_, data_ + target_count, origin_addr);
306 }
307 }

402 if (has_env) {
